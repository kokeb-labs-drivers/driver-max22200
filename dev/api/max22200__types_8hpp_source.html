608constexpr uint32_t EncodeChannel(const ChannelConfig& c) noexcept {
609 uint32_t v = 0;
610 v |= c.half_scale ? reg::kHfs : 0U;
611 v |= (static_cast<uint32_t>(c.hold) & reg::kCodeMask) << reg::kHoldShift;
612 v |= c.trigger_pin ? reg::kTrgnspi : 0U;
613 v |= (static_cast<uint32_t>(c.hit) & reg::kCodeMask) << reg::kHitShift;
614 v |= static_cast<uint32_t>(c.hit_time) << reg::kHitTimeShift;
615 v |= (c.drive == DriveMode::Vdr) ? reg::kVdrncdr : 0U;
616 v |= (c.side == Side::High) ? reg::kHsnls : 0U;
617 v |= (static_cast<uint32_t>(detail::ToUnderlying(c.chop)) & 0x03U) << reg::kFreqCfgShift;
618 v |= c.slew_rate_control ? reg::kSrc : 0U;
619 v |= c.open_load_detect ? reg::kOlEn : 0U;
620 v |= c.plunger_detect ? reg::kDpmEn : 0U;
621 v |= c.hit_current_check ? reg::kHhfEn : 0U;
622 return v;
623}

630constexpr ChannelConfig DecodeChannel(uint32_t v) noexcept {
631 ChannelConfig c{};
632 c.half_scale = (v & reg::kHfs) != 0U;
633 c.hold = static_cast<uint8_t>((v >> reg::kHoldShift) & reg::kCodeMask);
634 c.trigger_pin = (v & reg::kTrgnspi) != 0U;
635 c.hit = static_cast<uint8_t>((v >> reg::kHitShift) & reg::kCodeMask);
636 c.hit_time = static_cast<uint8_t>((v >> reg::kHitTimeShift) & 0xFFU);
637 c.drive = ((v & reg::kVdrncdr) != 0U) ? DriveMode::Vdr : DriveMode::Cdr;
638 c.side = ((v & reg::kHsnls) != 0U) ? Side::High : Side::Low;
639 c.chop = static_cast<ChopDivider>((v >> reg::kFreqCfgShift) & 0x03U); // 0 to 3: all enumerators
640 c.slew_rate_control = (v & reg::kSrc) != 0U;
641 c.open_load_detect = (v & reg::kOlEn) != 0U;
642 c.plunger_detect = (v & reg::kDpmEn) != 0U;
643 c.hit_current_check = (v & reg::kHhfEn) != 0U;
644 return c;

655 const FaultMask& m = config.fault_mask;
656 uint32_t v = static_cast<uint32_t>(on_mask) << reg::kOnchShift;
657 v |= m.overtemperature ? reg::kMaskOvt : 0U;
658 v |= m.overcurrent ? reg::kMaskOcp : 0U;
659 v |= m.open_load ? reg::kMaskOlf : 0U;
660 v |= m.hit_not_reached ? reg::kMaskHhf : 0U;
661 v |= m.plunger ? reg::kMaskDpm : 0U;
662 v |= m.communication ? reg::kMaskComf : 0U;
663 v |= m.undervoltage ? reg::kMaskUvm : 0U;
664 v |= (config.clock == MasterClock::Khz80) ? reg::kFreqm : 0U;
665 for (uint8_t p = 0; p < kPairs; ++p) {
666 v |= (static_cast<uint32_t>(detail::ToUnderlying(config.pairs[p])) & 0x03U)
