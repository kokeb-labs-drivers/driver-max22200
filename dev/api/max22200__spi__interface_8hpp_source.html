165constexpr std::array<uint8_t, 4> LsbFirst(uint32_t v) noexcept {
166 return {static_cast<uint8_t>(v & 0xFFU), static_cast<uint8_t>((v >> 8U) & 0xFFU),
167 static_cast<uint8_t>((v >> 16U) & 0xFFU), static_cast<uint8_t>(v >> 24U)};

306 Error command(uint8_t cmd, Reply& reply, bool guard) noexcept {
307 reply = Reply{};
308 if (!port_.GpioSet(CtrlPin::Cmd, GpioSignal::ACTIVE)) {
309 return Error::Pin;
310 }
311 const std::array<uint8_t, 1> tx = {cmd};
312 std::array<uint8_t, 1> rx = {0xFF};
313 if (!port_.Transfer(tx.data(), rx.data(), rx.size())) {
314 return Error::Bus;
315 }
316 take(rx[0], reply);
317 return (guard && ResetByte(rx[0])) ? Error::DeviceReset : Error::Ok;
318 }

321 static void take(uint8_t status, Reply& reply) noexcept {
322 reply.status = status;
323 if (status != 0xFFU) {
324 reply.flags =
325 static_cast<uint8_t>(static_cast<uint32_t>(reply.flags) | static_cast<uint32_t>(status));
326 }

336 if (!port_.GpioSet(CtrlPin::Cmd, GpioSignal::INACTIVE)) {
337 (void)port_.GpioSet(CtrlPin::Cmd, GpioSignal::ACTIVE);
338 return Error::Pin;

341 const bool high = port_.GpioSet(CtrlPin::Cmd, GpioSignal::ACTIVE);
342 if (!sent) {
343 return Error::Bus;
344 }
