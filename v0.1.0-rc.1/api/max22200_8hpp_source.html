31// Optional hooks: a `Declares*` concept tests that the port has a member of that name (a single
32// member by its address, an overload set by a call with the documented arguments), the `Has*`
33// concept the exact signature, `noexcept` and the result type; the driver rejects a hook that is
34// declared but does not match (KB-CPP20-007).
35
37template <typename Port>
38concept DeclaresGpioRead = requires {
39 &Port::GpioRead;
40} || requires(Port& port, GpioSignal& signal) { port.GpioRead(CtrlPin::Fault, signal); };
41
45template <typename Port>
46concept HasGpioRead =
47 requires { ExactHook<Port, bool(CtrlPin, GpioSignal&) noexcept>::Take(&Port::GpioRead); };
48
50template <typename Port>
51concept DeclaresDelayUs =
52 requires { &Port::DelayUs; } || requires(Port& port) { port.DelayUs(1U); };
53
56template <typename Port>
57concept HasDelayUs = requires { ExactHook<Port, void(uint32_t) noexcept>::Take(&Port::DelayUs); };
58
60template <typename Port>
61concept DeclaresLog =
62 requires { &Port::Log; } || requires(Port& port) { port.Log(LogLevel::Info, ""); };
63
65template <typename Port>
66concept HasLog =
67 requires { ExactHook<Port, void(LogLevel, const char*) noexcept>::Take(&Port::Log); };
68
71template <typename Port>
72void DelayUs(Port& port, uint32_t us) noexcept {
73 if constexpr (HasDelayUs<Port>) {
74 port.DelayUs(us);
75 } else {
76 static_assert(kAlwaysFalse<Port>,
77 "max22200: the port must implement `void DelayUs(uint32_t us) noexcept`");
78 }
79}

82template <typename Port>
83void Log(Port& port, [[maybe_unused]] LogLevel level,
84 [[maybe_unused]] const char* message) noexcept {
85 if constexpr (HasLog<Port>) {
86 port.Log(level, message);
87 } else {
88 (void)port;
89 }
90}

93constexpr uint8_t PairBits(uint8_t pair) noexcept {
94 return static_cast<uint8_t>(0x03U << (2U * (static_cast<uint32_t>(pair) & 0x03U)));
95}

98constexpr uint8_t ChannelBit(uint8_t channel) noexcept {
99 return static_cast<uint8_t>(uint32_t{1} << (static_cast<uint32_t>(channel) & 0x07U));
100}

102static_assert((PairBits(3_u8) == 0xC0U) && (ChannelBit(7_u8) == 0x80U),
103 "Table 10: ONCH7 is bit 31");
